! frr version 10.5.0
frr defaults datacenter
hostname R103
log syslog informational
service integrated-vtysh-config
! --- INTERFACCE ---
interface eth1
 description Link_to_R101_Internal
 ip address 10.0.0.6/30
exit
interface eth2
 description Link_to_R102_Internal
 ip address 10.0.0.10/30
exit
interface eth3
 description Link_to_R201_AS200
 ip address 203.0.113.9/30
exit
interface lo
 description Loopback_Router_ID
 ip address 3.3.3.3/32
exit
! --- BGP ---
router bgp 100
 bgp router-id 3.3.3.3
 neighbor 1.1.1.1 remote-as 100
 neighbor 1.1.1.1 update-source lo
 neighbor 2.2.2.2 remote-as 100
 neighbor 2.2.2.2 update-source lo
 ! remote-as 200 (diverso da AS 100) crea una sessione eBGP. 
 ! A differenza di iBGP, qui non usiamo "update-source lo" perché il peering avviene direttamente sugli IP fisici
 neighbor 203.0.113.10 remote-as 200
 address-family ipv4 unicast
  neighbor 1.1.1.1 next-hop-self
  neighbor 2.2.2.2 next-hop-self
 exit-address-family
exit
! --- OSPF ---
router ospf
 ospf router-id 3.3.3.3
 network 3.3.3.3/32 area 0
 network 10.0.0.4/30 area 0
 network 10.0.0.8/30 area 0
exit
